    <div class="card">
        <h1>ZPO Trainer</h1>
        <p>Unterkategorie:</p>
        <h2 id="frage">Lade Daten...</h2>
        
        <input type="text" id="inputUeberthema" placeholder="Überthema eingeben...">
        <input type="text" id="inputArtikel" placeholder="Artikel eingeben..." disabled>
        
        <div id="timer">0.00s</div>
        <div id="msg"></div>

        <div id="score-list">
            <h3>Letzte Scores:</h3>
            <ul id="scores"></ul>
        </div>
    </div>

    <script>
        // Beispiel-Daten (Hier kannst du deine ganze Liste einfügen)
        const daten = [
            { u: "Gegenstand und Geltungsbereich", t: "Allgemeine Bestimmungen", a: "1" },
            { u: "Schlichtungsversuch", t: "Besondere Bestimmungen", a: "197" },
            { u: "Mediation", t: "Besondere Bestimmungen", a: "213" }
        ];

        let aktuelleFrage, startTime, timerInterval;

        function neueFrage() {
            aktuelleFrage = daten[Math.floor(Math.random() * daten.length)];
            document.getElementById('frage').innerText = aktuelleFrage.u;
            document.getElementById('inputUeberthema').value = "";
            document.getElementById('inputUeberthema').disabled = false;
            document.getElementById('inputUeberthema').className = "";
            document.getElementById('inputArtikel').value = "";
            document.getElementById('inputArtikel').disabled = true;
            document.getElementById('timer').innerText = "0.00s";
            clearInterval(timerInterval);
        }

        document.getElementById('inputUeberthema').addEventListener('input', function(e) {
            if (e.target.value.trim().toLowerCase() === aktuelleFrage.t.toLowerCase()) {
                e.target.disabled = true;
                e.target.className = "correct";
                document.getElementById('inputArtikel').disabled = false;
                document.getElementById('inputArtikel').focus();
                startTimer();
            }
        });

        document.getElementById('inputArtikel').addEventListener('input', function(e) {
            if (e.target.value.trim() === aktuelleFrage.a) {
                stopTimer();
                e.target.className = "correct";
                setTimeout(neueFrage, 1500);
            }
        });

        function startTimer() {
            startTime = performance.now();
            timerInterval = setInterval(() => {
                const diff = (performance.now() - startTime) / 1000;
                document.getElementById('timer').innerText = diff.toFixed(2) + "s";
            }, 50);
        }

        function stopTimer() {
            clearInterval(timerInterval);
            const finalTime = ((performance.now() - startTime) / 1000).toFixed(2);
            const li = document.createElement('li');
            li.innerText = `${aktuelleFrage.u}: ${finalTime}s`;
            document.getElementById('scores').prepend(li);
        }

        neueFrage();
    </script>
